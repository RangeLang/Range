enum LoadError {
    case missing
    case network
}

construct User {
    @let id: Int
    @let name: String
}

function fetchUser(id: Int): Result<User, LoadError> {
    if id == 0 {
        return .failure(cause: .missing)
    }

    return .success(result: User(id: id, name: "George"))
}

function load<Value, Failure>(
    target: binding Promise<Value, Failure>,
    @autoclosure task: Result<Value, Failure>
) {
    state target: .loading

    @background {
        switch task() {
        case .success(let item):
            state target: .success(result: item)

        case .failure(let reason):
            state target: .failure(cause: reason)
        }
    }
}

construct UserView {
    @let id: Int
    @state user: Promise<User, LoadError>(.loading)

    init(id: Int) {
        state self.id: id
        load($self.user, fetchUser(id: id))
    }

    function reload() {
        load($self.user, fetchUser(id: id))
    }
}

@main {
    let view: UserView(id: 1)
}
