macro forwardOnlyArgument(): Init { target, diagnostics in
    if target.application.arguments.count == 1 {
        target.application.replace(replacement:
            target.application.arguments[0].expression
        )
    }
}

construct SingleInitRewriteFixture {
    @let number: Int

    @forwardOnlyArgument
    init(number: Int) {
        state self.number: number
    }
}

@main {
    SingleInitRewriteFixture(1)
}
