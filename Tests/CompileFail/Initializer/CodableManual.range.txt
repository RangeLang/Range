construct CodableManualUser {
    @let id: Int
    @let name: String
}

extension CodableManualUser {
    function encode(encoder: Encoder): Result<Void, EncodingError> {
        state container: encoder.keyedContainer()
        container.encode(id, key: "id")
        return container.encode(name, key: "name")
    }

    init(decoder: Decoder) -> Result<Self, DecodingError> {
        let container: decoder.keyedContainer()

        switch container.decode(Int.self, key: "id") {
        case .success(let id):
            state self.id: id
        case .failure(let error):
            return .failure(cause: error)
        }

        switch container.decode(String.self, key: "name") {
        case .success(let name):
            state self.name: name
        case .failure(let error):
            return .failure(cause: error)
        }

        return .success(result: self)
    }
}

function roundTripCodableManualUser(): Result<CodableManualUser, DecodingError> {
    let fields: [
        JSONField(key: "id", value: .int(value: 42)),
        JSONField(key: "name", value: .string(value: "George")),
    ]
    let decoder: JSONDecoder(value: .object(fields: fields))
    return CodableManualUser.decode(value: decoder)
}

@main {
    switch roundTripCodableManualUser() {
    case .success(let user):
        Logger.info(user.name)
    case .failure:
        Logger.error("decoding failed")
    }
}
