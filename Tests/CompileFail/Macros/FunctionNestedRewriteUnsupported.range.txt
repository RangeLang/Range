macro forwardFirstArgument(): Function { target, diagnostics in
    if !target.call.arguments.isEmpty {
        target.call.arguments[0].expression.replace(replacement:
            target.call.arguments[0].expression
        )
    }
}

@forwardFirstArgument
function passthrough(value: Int): Int {
    return value
}

@main {
    let output: Int(passthrough(value: 10))
}
