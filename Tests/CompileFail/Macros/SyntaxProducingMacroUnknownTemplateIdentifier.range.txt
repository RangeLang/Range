macro invalidEncodeSwitch(value: Identifier, key: String) -> Switch {
    switch container.encode(#(value), key: #(key)) {
    case .success:
        break
    case .failure(let error):
        return .failure(cause: error)
    }
}

macro emitInvalidEncodingSwitch(): Construct { target, diagnostics in
    target.declaration.expand {
        extension #(target.declaration.self) {
            function encode(encoder: Encoder): Result<Void, EncodingError> {
                let container: KeyedEncodingContainer(encoder.keyedContainer())
                @invalidEncodeSwitch(
                    value: Identifier(name: "id"),
                    key: "id"
                )
                return .success(result: Void())
            }
        }
    }
}

@emitInvalidEncodingSwitch
construct InvalidSyntaxProducingMacroSwitchFixture {
    @let id: Int
}
