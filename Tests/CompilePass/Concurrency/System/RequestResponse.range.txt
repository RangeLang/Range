construct Request {
    @let id: Int
    @let replyTo: Channel<String>
}

function fetchUserName(id: Int): String {
    if id == 0 {
        return "system"
    }

    return "george"
}

function server(requests: Channel<Request>) {
    @background {
        let first: Request(requests.receive())
        first.replyTo.send(fetchUserName(id: first.id))

        let second: Request(requests.receive())
        second.replyTo.send(fetchUserName(id: second.id))
    }
}

@main {
    let requests: Channel<Request>(storage: ChannelStorage<Request>(capacity: 2))
    let replyA: Channel<String>()
    let replyB: Channel<String>()
    let requestA: Request(id: 0, replyTo: replyA)
    let requestB: Request(id: 1, replyTo: replyB)

    server(requests: requests)

    requests.send(requestA)
    requests.send(requestB)

    Logger.info(replyA.receive())
    Logger.info(replyB.receive())
}
