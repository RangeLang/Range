macro firstArgument(): Function { target, diagnostics in
    if !target.call.arguments.isEmpty {
        target.call.replace(replacement:
            target.call.arguments[0].expression
        )
    }
}

@firstArgument
function select(text: String, number: Int): String {
    return text
}

@main {
    let result: Int(select(text: "Hello", number: 27))
}
