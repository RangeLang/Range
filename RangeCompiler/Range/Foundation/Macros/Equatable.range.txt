@macro(name: "compareProperty", result: "If")(lhs: T, rhs: T, value: Identifier){
    @generic(name: "T")

    if lhs.#(value) != rhs.#(value) {
        @return(value: false)
    }
}

@macro(name: "equatableBody", result: "Array<@statement>")(lhs: T, rhs: T, properties: Array<Let>){
    @generic(name: "T")

    #(properties.map { property in
        @compareProperty(lhs: lhs, rhs: rhs, value: property.identifier)
    })

    @return(value: true)
}

@construct
@macro(name: "equatable") { target, diagnostics in
    target.declaration.expand {
        extension #(target.declaration.self) {
            function ==(lhs: Self, rhs: Self): Bool {
                @equatableBody(
                    lhs: lhs,
                    rhs: rhs,
                    properties: target.declaration.lets
                )
            }
        }
    }
}
