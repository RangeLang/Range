@macro(name: "encodeProperty", result: "Switch")(container: KeyedEncodingContainer, value: Identifier, key: String){
    switch container.encode(#(value), key: key) {
    case .success:
        break
    case .failure(let error):
        @return(value: .failure(cause: error))
    }
}

@macro(name: "encodeBody", result: "Array<Switch>")(container: KeyedEncodingContainer, properties: Array<Let>){
    #(properties.map { property in
        let codingMacros: Array<Macro>(
            property.macros.filter { application in
            application.identifier.name == "codable"
            }
        )
        let behaviors: Array<CodingBehavior>(
            codingMacros.map { application in
            application.value
            }
        )
        let keys: Array<String>(
            behaviors.map { behavior in
            behavior.key ?? property.identifier.name
            }
        )
        let key: keys.first(default: property.identifier.name)

        @encodeProperty(
            container: container,
            value: property.identifier,
            key: key
        )
    })
}

@macro(name: "decodeProperty", result: "Switch")(container: KeyedDecodingContainer, type: TypeReference, value: Identifier, key: String){
    switch container.decode(#(type).self, key: key) {
    case .success(let #(value)):
        state self.#(value): #(value)
    case .failure(let error):
        @return(value: .failure(cause: error))
    }
}

@macro(name: "decodeBody", result: "Array<Switch>")(container: KeyedDecodingContainer, properties: Array<Let>){
    #(properties.map { property in
        let codingMacros: Array<Macro>(
            property.macros.filter { application in
            application.identifier.name == "codable"
            }
        )
        let behaviors: Array<CodingBehavior>(
            codingMacros.map { application in
            application.value
            }
        )
        let keys: Array<String>(
            behaviors.map { behavior in
            behavior.key ?? property.identifier.name
            }
        )
        let key: keys.first(default: property.identifier.name)

        @decodeProperty(
            container: container,
            type: property.type,
            value: property.identifier,
            key: key
        )
    })
}

@construct
@macro(name: "codable") {
    @generic(name: "shape") {
        @value(type: "CodingShape", current: "CodingShape.object")
    }
    target, diagnostics in
    @if(shape != .object) {
        diagnostics.error("@codable currently supports object coding only.")
    }

    @let(
        name: "properties",
        value: target.declaration.lets.filter { property in
            @let(
                name: "codingMacros",
                value: property.macros.filter { application in
                    @return(value: application.identifier.name == "codable")
                }
            )
            @let(
                name: "behaviors",
                value: codingMacros.map { application in
                    @return(value: application.value)
                }
            )
            @let(
                name: "excluded",
                value: behaviors.filter { behavior in
                    @return(value: behavior.exclude)
                }
            )
            @return(value: excluded.count == 0)
        }
    )

    target.declaration.expand {
        extension #(target.declaration.self) {
            function encode(encoder: Encoder): Result<Void, EncodingError> {
                let container: KeyedEncodingContainer(encoder.keyedContainer())
                @encodeBody(
                    container: container,
                    properties: properties
                )
                @return(value: .success(result: Void()))
            }

            function decode(decoder: Decoder): Result<Self, DecodingError> {
                let container: KeyedDecodingContainer(decoder.keyedContainer())
                @decodeBody(
                    container: container,
                    properties: properties
                )
                @return(value: .success(result: self))
            }
        }
    }
}
