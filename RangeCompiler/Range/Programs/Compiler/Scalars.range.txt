construct ScalarApplication {
    @let macroName: String
    @let targetName: String
    @let path: String
    @let source: String
}

construct ScalarDeclaration {
    @let macroName: String
    @let targetName: String
    @let llvmType: String
    @let path: String
}

function scalarBuild(): String {
    let files: Array<ProgramSourceFile>(scalarSourceFiles())
    let applications: Array<ScalarApplication>(collectScalarApplications(files: files))
    let declarations: Array<ScalarDeclaration>(resolveScalarDeclarations(applications: applications))

    state output: String("# Range scalar build file\n")

    for declaration in declarations {
        output: String(output + scalarDeclarationLine(declaration: declaration))
    }

    return output
}

function scalarSourceFiles(): Array<ProgramSourceFile> {
    let project: RangeProject(path: "RangeCompiler/Range/Core/DataSystem")
    return project.sourceFiles()
}

function collectScalarApplications(files: Array<ProgramSourceFile>): Array<ScalarApplication> {
    state applications: Array<ScalarApplication>([])

    for file in files {
        let targetName: String(constructName(path: file.path))

        if file.text.contains(value: "@int") {
            applications.append(
                element: ScalarApplication(
                    macroName: "int",
                    targetName: targetName,
                    path: file.path,
                    source: file.text
                )
            )
        }

        if file.text.contains(value: "@bool") {
            applications.append(
                element: ScalarApplication(
                    macroName: "bool",
                    targetName: targetName,
                    path: file.path,
                    source: file.text
                )
            )
        }

        if file.text.contains(value: "@boolean") {
            applications.append(
                element: ScalarApplication(
                    macroName: "boolean",
                    targetName: targetName,
                    path: file.path,
                    source: file.text
                )
            )
        }

        if file.text.contains(value: "@float") {
            applications.append(
                element: ScalarApplication(
                    macroName: "float",
                    targetName: targetName,
                    path: file.path,
                    source: file.text
                )
            )
        }
    }

    return applications
}

function resolveScalarDeclarations(applications: Array<ScalarApplication>): Array<ScalarDeclaration> {
    state declarations: Array<ScalarDeclaration>([])

    for application in applications {
        if scalarApplicationTargetsConstruct(application: application) {
            declarations.append(element: scalarDeclaration(application: application))
        }
    }

    return declarations
}

function scalarApplicationTargetsConstruct(application: ScalarApplication): Bool {
    if application.macroName == "int" {
        return application.source.contains(value: "@int\nconstruct")
    }

    if application.macroName == "bool" {
        return application.source.contains(value: "@bool\nconstruct")
    }

    if application.macroName == "boolean" {
        return application.source.contains(value: "@boolean\nconstruct")
    }

    if application.macroName == "float" {
        return application.source.contains(value: "@float\nconstruct")
    }

    return false
}

function scalarDeclaration(application: ScalarApplication): ScalarDeclaration {
    return ScalarDeclaration(
        macroName: application.macroName,
        targetName: application.targetName,
        llvmType: scalarLLVMType(application: application),
        path: application.path
    )
}

function scalarLLVMType(application: ScalarApplication): String {
    if application.macroName == "int" {
        return String("i" + scalarGenericValue(source: application.source, name: "bits", defaultValue: "64"))
    }

    if application.macroName == "bool" {
        return "i1"
    }

    if application.macroName == "boolean" {
        return "i1"
    }

    if application.macroName == "float" {
        let precision: String(scalarGenericValue(source: application.source, name: "precision", defaultValue: "64"))

        if precision == "32" {
            return "float"
        }

        return "double"
    }

    return ""
}

function scalarGenericValue(source: String, name: String, defaultValue: String): String {
    let stringPattern: String(name + ": String(\"")

    if source.contains(value: stringPattern) {
        let afterPattern: String(source.segment(index: 1, delimiter: stringPattern))
        return afterPattern.segment(index: 0, delimiter: "\")")
    }

    return defaultValue
}

function scalarDeclarationLine(declaration: ScalarDeclaration): String {
    return String(
        "scalar\t"
            + declaration.macroName
            + "\t"
            + declaration.targetName
            + "\t"
            + declaration.llvmType
            + "\t"
            + declaration.path
            + "\n"
    )
}

function constructName(path: String): String {
    let slashCount: Int(path.count(of: "/"))
    let fileName: String(path.segment(index: slashCount, delimiter: "/"))
    return fileName.replacingOccurrences(of: ".range", with: "")
}
