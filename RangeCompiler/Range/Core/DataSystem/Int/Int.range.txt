@int
construct Int<bits: String("64"), signedness: Signedness.signed> {
    @let storage: Memory<self.identifier>

    function literal(literal: IntLiteral): Self {
        return Self(memory: literal.storage())
    }
}

extension Int {
    function ==(lhs: Self, rhs: Self): Bool
}

extension Int {
    function <(lhs: Self, rhs: Self): Bool
}

@addable
extension Int {
    function +(lhs: Self, rhs: Self): Self
}

extension Int {
    function -(lhs: Self, rhs: Self): Self
}

extension Int {
    function *(lhs: Self, rhs: Self): Self
}

extension Int {
    function /(lhs: Self, rhs: Self): Self
}

extension Int {
    function %(lhs: Self, rhs: Self): Self
}

extension Int {
    function decode(decoder: Decoder): Result<Self, DecodingError> {
        let container: decoder.singleValueContainer()
        switch container.decode(Int.self) {
        case .success(let value):
            return .success(result: value)
        case .failure(let error):
            return .failure(cause: error)
        }
    }
}

extension Int {
    function encode(encoder: Encoder): Result<Void, EncodingError> {
        let container: encoder.singleValueContainer()
        return container.encode(self)
    }
}

extension Int {
    function hash(hasher: Hasher) {
        hasher.combineStorage(storage)
    }
}
