construct DictionaryStorage<Key, Value> {
    derived count: Int
    derived isEmpty: Bool
    function value(key: Key): Optional<Value>
    function updateValue(value: Value, key: Key)
    function removeValue(key: Key): Optional<Value>
    function contains(key: Key): Bool
    function entries(): Array<DictionaryEntry<Key, Value>>
    function filter(include: (DictionaryEntry<Key, Value>) -> Bool): Array<DictionaryEntry<Key, Value>>
    function first(include: (DictionaryEntry<Key, Value>) -> Bool): Optional<DictionaryEntry<Key, Value>>
    function clear()
}
