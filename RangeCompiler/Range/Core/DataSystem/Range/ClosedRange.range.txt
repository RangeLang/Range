@equatable
@hashable
construct ClosedRange<Bound> {
    @let lowerBound: Bound
    @let upperBound: Bound

    function contains(value: Bound): Bool {
        return !(value < lowerBound) && !(upperBound < value)
    }

    function overlaps(other: ClosedRange<Bound>): Bool {
        return contains(other.lowerBound) || other.contains(lowerBound)
    }

    derived isEmpty: Bool {
        return upperBound < lowerBound
    }
}
