construct Hasher {
    @state storage: HasherStorage()

    function combine<T>(value: T) {
        value.hash(hasher: self)
    }

    function combineStorage(value: IntStorage) {
        storage.combine(value)
    }

    function combineStorage(value: StringStorage) {
        storage.combine(value)
    }

    function combineStorage(value: BoolStorage) {
        storage.combine(value)
    }

    function combineStorage(value: DateStorage) {
        storage.combine(value)
    }

    function combineStorage(value: DateTimeStorage) {
        storage.combine(value)
    }

    function combineStorage(value: FloatStorage) {
        storage.combine(value)
    }

    function combineStorage(value: DataStorage) {
        storage.combine(value)
    }

    function combineStorage(value: UUIDStorage) {
        storage.combine(value)
    }

    function finalize(): Int {
        return storage.finalize()
    }
}
