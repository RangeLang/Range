@array
@literal
@syntax
construct Array<Element> {
    @state storage: ArrayStorage<Element>()

    derived count: Int {
        storage.count
    }

    derived isEmpty: Bool {
        storage.isEmpty
    }

    function append(element: Element) {
        storage.append(element: element)
    }

    function element(index: Int): Element {
        return storage.element(index: index)
    }

    function first(): Optional<Element> {
        return storage.first()
    }

    function first(include: (Element) -> Bool): Optional<Element> {
        return storage.first(include: include)
    }

    function last(): Optional<Element> {
        return storage.last()
    }

    function update(element: Element, index: Int) {
        storage.update(element: element, index: index)
    }

    function insert(element: Element, index: Int) {
        storage.insert(element: element, index: index)
    }

    function remove(index: Int): Element {
        return storage.remove(index: index)
    }

    function removeLast(): Optional<Element> {
        return storage.removeLast()
    }

    function clear() {
        storage.clear()
    }

    function filter(include: (Element) -> Bool): Array<Element> {
        return Array(storage: storage.filter(include: include))
    }

    function map<Output>(transform: (Element) -> Output): Array<Output> {
        return Array<Output>(storage: storage.map(transform))
    }

    function compactMap<Output>(transform: (Element) -> Optional<Output>): Array<Output> {
        return Array<Output>(storage: storage.compactMap(transform))
    }

    function flatMap<Output>(transform: (Element) -> Array<Output>): Array<Output> {
        return Array<Output>(storage: storage.flatMap(transform))
    }

    function forEach(body: (Element) -> Void) {
        storage.forEach(body)
    }
}

extension Array {
    function encode(encoder: Encoder): Result<Void, EncodingError> {
        state container: encoder.unkeyedContainer()
        state index: 0

        while index < count {
            switch container.encode(element(index: index)) {
            case .success:
                state index: index + 1
            case .failure(let error):
                return .failure(cause: error)
            }
        }

        return .success(result: Void())
    }
}
