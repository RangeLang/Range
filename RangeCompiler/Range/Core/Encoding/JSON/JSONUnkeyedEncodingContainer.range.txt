construct JSONUnkeyedEncodingContainer {
    binding output: String
    binding isArray: Bool
    @state wroteValue: Bool(false)

    function encode(value: Bool): Result<Void, EncodingError> {
        appendValue(value: "\(value)")
        return .success(result: Void())
    }

    function encode(value: Data): Result<Void, EncodingError> {
        return .failure(cause: .failed)
    }

    function encode(value: Date): Result<Void, EncodingError> {
        appendValue(value: "\"\(value)\"")
        return .success(result: Void())
    }

    function encode(value: DateTime): Result<Void, EncodingError> {
        appendValue(value: "\"\(value)\"")
        return .success(result: Void())
    }

    function encode(value: Float): Result<Void, EncodingError> {
        appendValue(value: "\(value)")
        return .success(result: Void())
    }

    function encode(value: Int): Result<Void, EncodingError> {
        appendValue(value: "\(value)")
        return .success(result: Void())
    }

    function encode(value: String): Result<Void, EncodingError> {
        appendValue(value: "\"\(value)\"")
        return .success(result: Void())
    }

    function encode(value: UUID): Result<Void, EncodingError> {
        appendValue(value: "\"\(value)\"")
        return .success(result: Void())
    }

    function encode<T>(value: T): Result<Void, EncodingError> {
        return .failure(cause: .failed)
    }

    function appendValue(value: String) {
        if wroteValue {
            state output: output + ","
        } else {
            state wroteValue: true
        }

        state output: output + value
    }
}
