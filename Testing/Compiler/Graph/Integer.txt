Macro {
    macros: {
        Attribute {
            name: literal
            arguments: {
                Argument {
                    value: String { value: "[0-9]+" }
                }
            }
        }
    }
    name: integer
    target: Construct
    body: {
        members: {
            value: #target.members.filter(named: "value").first
            bits: #target.generics.filter(named: "bits").first
            signed: #target.generics.filter(named: "signed").first
        }
    }
}

Function {
    name: integerValueFits
    parameters: {
        Parameter {
            name: value
            type: Name {
                name: Int
            }
        }
        Parameter {
            name: bits
            type: Name {
                name: Int
            }
        }
        Parameter {
            name: signed
            type: Name {
                name: Bool
            }
        }
    }
    output: Bool
    body: {
        If {
            condition: Binary {
                name: "<="
                left: Name {
                    name: bits
                }
                right: 0
            }
            then: {
                Return {
                    value: false
                }
            }
        }
        Local {
            name: remaining
            type: Int
            binding: state
            arguments: {
                Argument {
                    value: Name {
                        name: value
                    }
                }
            }
        }
        Local {
            name: available
            type: Int
            binding: state
            arguments: {
                Argument {
                    value: Name {
                        name: bits
                    }
                }
            }
        }
        If {
            condition: Name {
                name: signed
            }
            then: {
                Assign {
                    target: Name {
                        name: available
                    }
                    value: Binary {
                        name: "-"
                        left: Name {
                            name: available
                        }
                        right: 1
                    }
                }
                If {
                    condition: Binary {
                        name: "<"
                        left: Name {
                            name: remaining
                        }
                        right: 0
                    }
                    then: {
                        Assign {
                            target: Name {
                                name: remaining
                            }
                            value: Binary {
                                name: "-"
                                left: Unary {
                                    name: "-"
                                    value: 1
                                }
                                right: Name {
                                    name: remaining
                                }
                            }
                        }
                    }
                }
            }
            else: {
                If {
                    condition: Binary {
                        name: "<"
                        left: Name {
                            name: remaining
                        }
                        right: 0
                    }
                    then: {
                        Return {
                            value: false
                        }
                    }
                }
            }
        }
        While {
            condition: Binary {
                name: "&&"
                left: Binary {
                    name: ">"
                    left: Name {
                        name: remaining
                    }
                    right: 0
                }
                right: Binary {
                    name: ">"
                    left: Name {
                        name: available
                    }
                    right: 0
                }
            }
            body: {
                Assign {
                    target: Name {
                        name: remaining
                    }
                    value: Binary {
                        name: "/"
                        left: Name {
                            name: remaining
                        }
                        right: 2
                    }
                }
                Assign {
                    target: Name {
                        name: available
                    }
                    value: Binary {
                        name: "-"
                        left: Name {
                            name: available
                        }
                        right: 1
                    }
                }
            }
        }
        Return {
            value: Binary {
                name: "=="
                left: Name {
                    name: remaining
                }
                right: 0
            }
        }
    }
}
