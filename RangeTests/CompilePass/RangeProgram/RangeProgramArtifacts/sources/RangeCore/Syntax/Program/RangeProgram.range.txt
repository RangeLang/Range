#language
construct RangeProgram {
    let graph: ProgramGraph
}

#language
construct ProgramGraph {
    let files: [FileTreeNode]
    let sources: [ProgramSourceFile]
    let lexedSources: [ProgramLexedSource]
    let artifacts: [ProgramArtifact]
    let rangeGraph: RangeGraph
    let syntax: [GraphSyntax]
}

#language
construct ProgramSourceFile {
    let path: String
    let text: String
}

#language
construct ProgramLexedSource {
    let path: String
    let tokens: [LexicalToken]
}

#language
construct ProgramArtifact {
    let path: String
    let text: String
}

#language
construct RangeGraph {
    let nodes: [RangeGraphNode]
    let relationships: [RangeGraphRelationship]
}

#language
construct RangeGraphNode {
    let identity: RangeGraphIdentity
    let sourcePath: String
    let labels: [String]
}

#language
construct RangeGraphRelationship {
    let origin: RangeGraphIdentity
    let destination: RangeGraphIdentity
    let kind: String
}

#language
construct GraphSyntax {
    let identity: RangeGraphIdentity
    let declarations: [#GraphDeclaration]
    let applications: [#GraphApplication]
}

#language
construct RangeGraphIdentity {
    let id: String
    let kind: String
    let name: String
}

#language
construct GraphDeclaration {
    let identity: RangeGraphIdentity
}

#language
construct GraphApplication {
    let identity: RangeGraphIdentity
}
