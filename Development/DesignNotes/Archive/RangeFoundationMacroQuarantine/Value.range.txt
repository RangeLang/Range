@macro(name: "value", result: @string, target: @member){
    @parameter(name: "type", value: @string(value: ""))
    @parameter(name: "current", value: @string(value: ""))
    @parameter(name: "body", value: @string(value: ""))
    @return(value: "member|kind=value|type="
        + type
        + "|current="
        + current
        + "|ordinal="
        + String(target.ordinal)
        + "\n"
        + body)
}
