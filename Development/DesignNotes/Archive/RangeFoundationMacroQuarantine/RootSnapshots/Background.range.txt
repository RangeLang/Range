@macro(name: "block", result: "Array<@statement>")(statements: Array<@statement>){
    #(statements)
}

@block
@syntax
@macro(name: "background"){ target, diagnostics, graph in
    @if("target.declaration.statements.count == Int(0)") {
        diagnostics.error("@background block must not be empty.")
    }

    target.replace(replacement: {
        switch Thread.spawn({
            @block(statements: target.declaration.statements)
            Thread.yield()
        }) {
        case .success(let handle):
            Thread.detach(handle)
        case .failure:
            Logger.error("@background failed to spawn thread.")
        }
    })
}
