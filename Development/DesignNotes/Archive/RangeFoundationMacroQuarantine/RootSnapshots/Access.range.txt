            }
        )
    )
    @let(
        name: "accessibleMacros",
        value: Enum(
            self: NamedTypeReference(name: "AccessibleMacros"),
            generics: [],

                EnumCase(identifier: macro.identifier, associatedValues: [])
            }
        )
    )
    @let(
        name: "accessibleConstructs",

            #(accessibleMacros)
            #(accessibleConstructs)
        }
    }
