@macro(name: "assignment", result: @string, target: @statement){
    @parameter(name: "target", value: @string(value: ""))
    @parameter(name: "value", value: @string(value: ""))

    @return(value: "statement|kind=assign|target="
        + target
        + "|value="
        + value
        + "|projection="
        + "target.declaration")
}
